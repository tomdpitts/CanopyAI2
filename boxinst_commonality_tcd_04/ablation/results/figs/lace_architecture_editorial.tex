\newcommand{\stage}[5]{%
  \node[inner sep=0pt, anchor=base west, font=\fontsize{7.6}{9}\selectfont\addfontfeature{LetterSpace=12}\bfseries, text=ink]

  \draw[ink, line width=0.5pt] (#1,{#2-0.14}) -- ++(#4,0);
